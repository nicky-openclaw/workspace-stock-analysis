# B2+砖型图 组合选股公式
#
# 公式说明：
# - B2: 昨日J值负 + 涨幅>3.95% + 放量
# - 砖型图: 红砖转绿砖 + 砖型值上升
#
# 使用方法：
# 1. 打开通达信软件
# 2. 功能 -> 公式管理器 -> 新建
# 3. 粘贴此公式
# 4. 设置参数和选股条件

# ==================== 砖型图部分 ====================
# 4日周期最高
H4 := HHV(H, 4);
# 4日周期最低
L4 := LLV(L, 4);

# 砖型图计算
VAR1A := ((H4 - C) / (H4 - L4)) * 100 - 90;
VAR3A := ((C - L4) / (H4 - L4)) * 100;
VAR6A := VAR3A - VAR1A;
砖型图 := IF(VAR6A > 4, VAR6A - 4, 0);

# 昨日砖型图
VAR1A_1 := ((REF(H4,1) - REF(C,1)) / (REF(H4,1) - REF(L4,1))) * 100 - 90;
VAR3A_1 := ((REF(C,1) - REF(L4,1)) / (REF(H4,1) - REF(L4,1))) * 100;
VAR6A_1 := VAR3A_1 - VAR1A_1;
昨日砖型图 := IF(VAR6A_1 > 4, VAR6A_1 - 4, 0);

# 砖型图走强：今日砖型 > 昨日砖型
砖型走强 := 砖型图 > 昨日砖型图 AND 砖型图 > 0;

# 砖型图红砖：收盘 > 开盘
红砖 := C > O;

# 昨日绿砖：昨日收盘 < 昨日开盘
昨日绿砖 := REF(C,1) < REF(O,1);

# ==================== B2部分 ====================
# KDJ计算
N := 9;
M1 := 3;
M2 := 3;

RSV := (C - LLV(L,N)) / (HHV(H,N) - LLV(L,N)) * 100);
K := SMA(RSV, M1, 1);
D := SMA(K, M2, 1);
J := 3*K - 2*D;

# 昨日J值
昨日J := REF(J,1);

# 涨幅
涨幅 := (C - REF(C,1)) / REF(C,1) * 100;

# 放量：今日量 > 昨日量 * 1.5
放量 := V > REF(V,1) * 1.5;

# B2条件
B2信号 := 昨日J < 0 AND 涨幅 > 3.95 AND 放量;

# ==================== 组合条件 ====================
# B2 + 砖型图共振
选股条件 := B2信号 AND 砖型走强 AND 红砖 AND 昨日绿砖;

# 输出信号
信号 := 选股条件;

# ==================== 选股条件说明 ====================
# 1. 砖型图走强：今日砖型值 > 昨日砖型值
# 2. 红砖：今日收盘 > 开盘
# 3. 昨日绿砖：昨日收盘 < 开盘
# 4. B2信号：昨日J<0 + 涨幅>3.95% + 放量>1.5倍
#
# 同时满足以上条件时，选股信号触发
